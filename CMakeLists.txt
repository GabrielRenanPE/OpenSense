cmake_minimum_required(VERSION 3.16)
project(OpenSense VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
include(GNUInstallDirs)
configure_file(polkit/org.opensense.cooling.policy.in
               org.opensense.cooling.policy @ONLY)
add_executable(opensense src/main.cpp)
target_link_libraries(opensense PRIVATE Qt6::Widgets)
add_executable(opensense-helper src/helper.cpp)
add_executable(opensense-fan-helper src/helper.cpp)
target_compile_definitions(opensense-fan-helper PRIVATE OPENSENSE_FAN_ONLY=1)
target_compile_options(opensense-fan-helper PRIVATE -Wall -Wextra)
target_compile_options(opensense PRIVATE -Wall -Wextra)
target_compile_options(opensense-helper PRIVATE -Wall -Wextra)

# Hardening dos binários instalados (os auxiliares rodam como root via pkexec).
# Cada opção só é usada se o compilador/ligador a suportar.
include(CheckCXXCompilerFlag)
include(CheckPIESupported)
check_pie_supported(LANGUAGES CXX)
check_cxx_compiler_flag(-fstack-protector-strong OPENSENSE_HAS_STACK_PROTECTOR)
check_cxx_compiler_flag(-Wl,-z,relro,-z,now OPENSENSE_HAS_RELRO_NOW)
foreach(hardened IN ITEMS opensense opensense-helper opensense-fan-helper)
  target_compile_definitions(${hardened} PRIVATE _GLIBCXX_ASSERTIONS)
  if(OPENSENSE_HAS_STACK_PROTECTOR)
    target_compile_options(${hardened} PRIVATE -fstack-protector-strong)
  endif()
  if(OPENSENSE_HAS_RELRO_NOW)
    target_link_options(${hardened} PRIVATE -Wl,-z,relro,-z,now)
  endif()
  if(CMAKE_CXX_LINK_PIE_SUPPORTED)
    set_target_properties(${hardened} PROPERTIES POSITION_INDEPENDENT_CODE ON)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 add_executable(core-tests tests/core.cpp)
 target_compile_options(core-tests PRIVATE -Wall -Wextra)
 add_test(NAME core COMMAND core-tests)
 add_test(NAME cooling-scope COMMAND ${CMAKE_COMMAND}
          -DHELPER=$<TARGET_FILE:opensense-fan-helper>
          -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cooling-scope.cmake)
 add_executable(ui-tests tests/ui.cpp)
 target_link_libraries(ui-tests PRIVATE Qt6::Widgets Qt6::Test)
 add_test(NAME ui COMMAND ui-tests)
 set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
install(TARGETS opensense RUNTIME DESTINATION bin)
install(TARGETS opensense-helper RUNTIME DESTINATION libexec)
install(TARGETS opensense-fan-helper RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/org.opensense.cooling.policy"
        DESTINATION /usr/share/polkit-1/actions)
install(FILES opensense.desktop DESTINATION share/applications)
install(FILES opensense.svg DESTINATION share/icons/hicolor/scalable/apps)
